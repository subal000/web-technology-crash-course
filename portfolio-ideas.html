Medium

Typing effect for your tagline

Make the tagline type itself out and cycle through phrases.

🔍 search: typing effect javascript / typed.js
Peek at a simple version 
<!-- put <span id="typed"></span> in your HTML -->
const words = ["a web developer", "a problem solver", "a creator"];
let w = 0, c = 0;
const el = document.querySelector("#typed");
function type() {
  el.textContent = words[w].slice(0, c++);
  if (c > words[w].length) { c = 0; w = (w + 1) % words.length; setTimeout(type, 1200); }
  else setTimeout(type, 90);
}
type();
Medium

"Back to top" button

A button that appears when you scroll down and glides you back up.

Peek at the code 
const top = document.createElement("button");
top.textContent = "↑";
top.style.cssText = "position:fixed;bottom:20px;right:20px;padding:10px 14px;border:none;border-radius:50%;background:var(--accent);color:#fff;cursor:pointer;display:none";
document.body.appendChild(top);
top.onclick = () => window.scrollTo({ top: 0, behavior: "smooth" });
window.onscroll = () => top.style.display = window.scrollY > 400 ? "block" : "none";

async function loadRepos() {
  const res = await fetch("https://api.github.com/users/YOUR-USERNAME/repos");
  const repos = await res.json();
  repos.forEach(r => console.log(r.name, "→", r.html_url));
  // now build cards from them, just like the person/hero apps!
}
loadRepos();
